\begin{frame}[t]{02. Problem Statement: Core Challenges in AI Agent Systems}
\vspace{-0.30em}

% Top Subtitle Banner
\centering
\begin{tcolorbox}[
  enhanced, colback=lavenderDark!90, colframe=lavenderDark, arc=3pt,
  boxrule=0.5pt, width=0.98\linewidth, top=2.8pt, bottom=2.8pt, left=8pt, right=8pt,
  halign=center
]
  \textcolor{white}{\fontsize{6.8}{8.4}\selectfont\sffamily
  \textbf{Core Challenge:} Current autonomous AI agent platforms struggle with monolithic bottlenecks, sequential execution delays, proprietary tool silos, and ungrounded execution risks when handling complex workflows.}
\end{tcolorbox}

\vspace{-0.35em}

% 2x2 Grid of Problem Cards
\begin{columns}[T]
  % Column 1
  \begin{column}{0.485\textwidth}
    % Card 1: Monolithic LLM Bottleneck
    \begin{tcolorbox}[
      enhanced, arc=3pt, colframe=red!65!black, colback=white,
      title={\scriptsize\bfseries\color{white}\faExclamationCircle\ 1. Monolithic LLM Bottleneck \& High Cost},
      colbacktitle=red!65!black, boxrule=0.6pt,
      left=4pt, right=4pt, top=2.0pt, bottom=2.0pt
    ]
    \fontsize{5.5}{6.7}\selectfont
    \begin{itemize}\setlength{\itemsep}{1.2pt}
      \item \textbf{Single-Model Overload:} One massive LLM attempts planning, tool use, and synthesis in a bloated prompt context.
      \item \textbf{Cost \& Latency Bloat:} Heavy frontier models incur $3\times$ higher token expense and slow turnaround.
      \item \textbf{\color{red!75!black}What We Solve:} Route sub-tasks dynamically to cost/latency-optimal models (Groq, OpenAI, NIM) with specialist agent roles.
    \end{itemize}
    \end{tcolorbox}

    \vspace{-0.30em}

    % Card 2: Sequential Execution & Fragility
    \begin{tcolorbox}[
      enhanced, arc=3pt, colframe=prismGold!85!black, colback=white,
      title={\scriptsize\bfseries\color{white}\faRandom\ 2. Sequential \& Rigid Task Execution},
      colbacktitle=prismGold!85!black, boxrule=0.6pt,
      left=4pt, right=4pt, top=2.0pt, bottom=2.0pt
    ]
    \fontsize{5.5}{6.7}\selectfont
    \begin{itemize}\setlength{\itemsep}{1.2pt}
      \item \textbf{No Parallel Concurrency:} Existing chains execute steps sequentially; independent tasks wait needlessly in line.
      \item \textbf{Brittle Pipeline Fragility:} A single sub-task failure crashes the entire linear pipeline with no recovery.
      \item \textbf{\color{prismGold!90!black}What We Solve:} Convert user goals into a dynamic DAG to execute independent tasks in parallel ($K \le 5$).
    \end{itemize}
    \end{tcolorbox}
  \end{column}

  % Column 2
  \begin{column}{0.485\textwidth}
    % Card 3: Tool Silos & Observability
    \begin{tcolorbox}[
      enhanced, arc=3pt, colframe=prismTeal!85!black, colback=white,
      title={\scriptsize\bfseries\color{white}\faTools\ 3. Fragmented Tool Silos \& Black-Box Runs},
      colbacktitle=prismTeal!85!black, boxrule=0.6pt,
      left=4pt, right=4pt, top=2.0pt, bottom=2.0pt
    ]
    \fontsize{5.5}{6.7}\selectfont
    \begin{itemize}\setlength{\itemsep}{1.2pt}
      \item \textbf{Proprietary Integrations:} Ad-hoc APIs make connecting daily tools (Gmail, Drive, Terminal, Local Files) difficult.
      \item \textbf{No Real-Time Feedback:} Users are left blind during long runs without streaming status or progress logs.
      \item \textbf{\color{prismTeal!90!black}What We Solve:} Standardize tool connectivity via Model Context Protocol (MCP) with live Server-Sent Events (SSE).
    \end{itemize}
    \end{tcolorbox}

    \vspace{-0.30em}

    % Card 4: Safety Risks & Cloud Lock-in
    \begin{tcolorbox}[
      enhanced, arc=3pt, colframe=lavenderDark!85!black, colback=white,
      title={\scriptsize\bfseries\color{white}\faLock\ 4. Safety Hazards \& Cloud Data Lock-in},
      colbacktitle=lavenderDark!85!black, boxrule=0.6pt,
      left=4pt, right=4pt, top=2.0pt, bottom=2.0pt
    ]
    \fontsize{5.5}{6.7}\selectfont
    \begin{itemize}\setlength{\itemsep}{1.2pt}
      \item \textbf{Unguarded Actions:} Agents execute irreversible actions (sending emails, deleting files) with zero human approval.
      \item \textbf{Privacy \& Latency Risks:} Session state stored in remote cloud databases exposes sensitive personal data.
      \item \textbf{\color{lavenderDark!90!black}What We Solve:} Human-in-the-loop approval, ORPO alignment, and zero-latency local browser database (\texttt{PrismDB}).
    \end{itemize}
    \end{tcolorbox}
  \end{column}
\end{columns}

\end{frame}
