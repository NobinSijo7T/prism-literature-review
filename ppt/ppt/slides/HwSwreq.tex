\section{Resources Required}
\begin{frame}{Resources Required}
    \begin{itemize}
        \item \textbf{Hardware \& Software Requirements}

            \begin{itemize}
                \item \textbf{Compute Resources:} Google Colab (GPU-accelerated) for data vectorization, processing, and indexing.

                \item \textbf{LLM Inference:} Groq Cloud for high-speed access to open-source Large Language Models.
            \end{itemize}
        \item \textbf{Databases \& Search:}

        \begin{itemize}
            % \item \textbf{Primary Database:}PGVector (PostgreSQL extension) for integrated relational and vector data storage.

            \item \textbf{Hybrid Search Engine:} BM25 for traditional keyword-based (lexical) retrieval.
            FAISS for efficient similarity searches and vector indexing.
        \end{itemize}
    \end{itemize}
\end{frame}
