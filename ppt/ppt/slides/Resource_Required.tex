\section{Resource Required}

    \begin{frame}{Resource Required}
        \small
        \textbf{Hardware & Software:}
        \begin{itemize}
            \item Cloud-based virtual machines (e.g., AWS EC2, Google Cloud Compute) for crawling and indexing.
        \end{itemize}
        
        \textbf{Databases:}
        \begin{itemize}
            \item Hybrid search engine (Elasticsearch/OpenSearch) for inverted index and vector search.
            \item Alternatively, vector database like Milvus or Pinecone.
        \end{itemize}
        
        \textbf{Libraries \& Frameworks:}
        \begin{itemize}
            \item Python Libraries: Scrapy, BeautifulSoup, Requests.
            \item Embedding Models: Hugging Face or OpenAI models.
        \end{itemize}
        
        \textbf{Front-End:}
        \begin{itemize}
            \item Modern JavaScript framework (\textbf{React}).
        \end{itemize}
    
    \end{frame}
