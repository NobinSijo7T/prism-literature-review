\intentcol{3}{cInt}{Intent Classifier}{Intent Classifier Agent}{(LLM / Classifier)}{%
  \intentbadge{cInt}{Function}
  Analyze contextual embeddings and curated supervision to infer user intent, extract domain entities (target API, destination notes), resolve dependencies, and evaluate confidence thresholds for task routing.\par\vspace{1.5pt}
  \intentbadge{cInt}{Input}
  Token embeddings + prompt context.\par\vspace{1.5pt}
  \intentbadge{cInt}{Output}
  Derived intent label(s) with confidence + entity slots; labels are inferred from the prompt and curated training targets, not assumed to be literal fields in every source record.}

\intentcol{5}{cDec}{Task DAG Output}{Tool Executor Agent}{(Tooling Agent)}{%
